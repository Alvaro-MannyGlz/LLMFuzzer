import java.util.Random;

public class RandomNumberGenerator {
    public static void main(String[] args) {
        // Initialize a Random object for generating random numbers
        Random randomGenerator = new Random();
        
        // Generate a random integer in the range [1, 100]
        int generatedNumber = randomGenerator.nextInt(100) + 1;
        
        // Output the generated number
        System.out.println("Random number between 1 and 100: " + generatedNumber);
    }
}
